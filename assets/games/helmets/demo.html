    helmetSVG('#cc2200', '#2244bb', '#cccccc');

// Server (Node.js) — store SVG strings in a database
const { helmetSVGString } = require('./helmet.js');
db.run('UPDATE teams SET helmet_svg = ? WHERE id = ?',
  helmetSVGString(team.shell, team.stripe, team.mask),
  team.id);
